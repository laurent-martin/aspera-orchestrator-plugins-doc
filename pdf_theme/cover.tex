\newcommand{\coverpx}[1]{\dimexpr\paperwidth*#1/1086\relax}
% Arguments: left, first baseline, width, text
% The text starts a paragraph: else a \color at its start would come before the first line, the box would be aligned
% on the color and not on the first baseline, which would be lower by the height of the line
\newcommand{\coverput}[4]{\AtPageUpperLeft{\hspace*{\coverpx{#1}}\raisebox{-\coverpx{#2}}{\parbox[t]{\coverpx{#3}}{\raggedright\leavevmode #4}}}}
% The picture is a PDF made from cover.svg, see README.md. It is 3:4, it goes past the bottom of a Letter page.
% cover.svg declares the size of a Letter page and stretches its 3:4 view box to it: stretched to 3:4 again, it has its shape
\renewcommand{\gfmcoverbackground}{\AddToShipoutPictureBG*{
    \AtPageUpperLeft{\raisebox{-\height}{\includegraphics[width=\paperwidth,height=\coverpx{1448}]{cover.pdf}}}
  }}

% Title of the document without the product name prefix, shown apart in bold
\ExplSyntaxOn
\NewDocumentCommand{\coverdocumenttitle}{}{
  \tl_set_eq:Nc \l_tmpa_tl {@title}
  % Expand \gfminputproductname into a tl, then append a space, then remove the prefix from the title
  \tl_set:Nv \l_tmpb_tl {gfminputproductname}
  \tl_put_right:Nn \l_tmpb_tl {~}
  \tl_remove_once:NV \l_tmpa_tl \l_tmpb_tl
  \tl_use:N \l_tmpa_tl
}
\ExplSyntaxOff
